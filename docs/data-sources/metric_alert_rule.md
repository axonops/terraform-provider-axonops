# axonops_metric_alert_rule (Data Source)

Reads a metric alert rule.

## Example Usage

```terraform
# Read an existing metric alert rule by ID
data "axonops_metric_alert_rule" "existing" {
  cluster_name = "my-cassandra-cluster"
  cluster_type = "cassandra"
  id           = "existing-alert-id"
}

# Output the alert rule details
output "alert_name" {
  value = data.axonops_metric_alert_rule.existing.name
}

output "alert_metric" {
  value = data.axonops_metric_alert_rule.existing.metric
}

output "alert_warning_threshold" {
  value = data.axonops_metric_alert_rule.existing.warning_value
}

output "alert_critical_threshold" {
  value = data.axonops_metric_alert_rule.existing.critical_value
}

output "alert_duration" {
  value = data.axonops_metric_alert_rule.existing.duration
}

output "alert_operator" {
  value = data.axonops_metric_alert_rule.existing.operator
}
```
