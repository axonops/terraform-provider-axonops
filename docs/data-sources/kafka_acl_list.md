# axonops_kafka_acl_list (Data Source)

Lists all Kafka ACLs for a cluster. Use this data source to retrieve the complete list of ACL entries configured for a Kafka cluster.

## Example Usage

```terraform
# List all ACLs for a cluster
data "axonops_kafka_acl_list" "all" {
  cluster_name = "my-kafka-cluster"
}

# Output the ACL list
output "acl_count" {
  value = length(data.axonops_kafka_acl_list.all.acls)
}

output "all_acls" {
  value = data.axonops_kafka_acl_list.all.acls
}

# Filter ACLs for a specific principal using locals
locals {
  producer_acls = [
    for acl in data.axonops_kafka_acl_list.all.acls :
    acl if acl.principal == "User:producer-app"
  ]
}

output "producer_acls" {
  value = local.producer_acls
}
```
